ESP32 DevKit V1

Microcontroller Core

Low-power 32-bit LX6 dual-core processor operating at 240MHz with built-in Wi-Fi 802.11 b/g/n and Bluetooth LE.

Operating Voltage: 3.3V - 5.0V
ADC Resolution: 12-bit (0-4095)
Protocols: HTTP / WebSockets

MQ135 Gas Sensor

Air Pollution Sensor

Uses Tin Dioxide (SnO2) semiconductor sensitive layer. Gas resistance decreases as concentration of harmful gases increases.

Target Gases: NH3, NOx, Alcohol, Smoke, CO2
Preheat Time: 24 Hours
Detection Range: 10 - 1000 PPM

DHT11 Climate Sensor

Temp & Humidity Sensor

Comprises a resistive-type humidity measurement component and an NTC temperature thermistor with a 1-wire digital communication bus.

Temp Range: 0 to 50 °C (±2°C)
Humidity Range: 20 - 90% RH (±5%)
Signal Pin: GPIO 4

ESP32 Arduino Firmware Code (C++) 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

//---------------- LCD ----------------//
LiquidCrystal_I2C lcd(0x27, 16, 2);

//---------------- DHT11 ----------------//
#define DHTPIN 32

#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

//---------------- MQ135 ----------------//
#define MQ135_PIN 34

//---------------- Output Pins ----------------//
#define BUZZER      18
#define RED_LED     19
#define GREEN_LED   25
#define YELLOW_LED  26

void setup()
{
  Serial.begin(115200);

  pinMode(BUZZER, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);

  digitalWrite(BUZZER, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);

  dht.begin();
  Wire.begin(21, 22);

  lcd.init();
  lcd.backlight();

  // Welcome Screen
  lcd.setCursor(5,0);
  lcd.print("AERO");
  lcd.setCursor(4,1);
  lcd.print("SENSE");
  delay(2000);

  // LED Test
  digitalWrite(BUZZER, HIGH);
  digitalWrite(RED_LED, HIGH);
  digitalWrite(GREEN_LED, HIGH);
  digitalWrite(YELLOW_LED, HIGH);
  delay(100);

  digitalWrite(BUZZER, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  delay(100);

  digitalWrite(BUZZER, HIGH);
  digitalWrite(RED_LED, HIGH);
  digitalWrite(GREEN_LED, HIGH);
  digitalWrite(YELLOW_LED, HIGH);
  delay(100);

  digitalWrite(BUZZER, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  delay(100);

  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("SMART AIR");
  lcd.setCursor(0,1);
  lcd.print("MONITOR SYSTEM");
  delay(2000);

  lcd.clear();
}

void loop()
{
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();
  int gas = analogRead(MQ135_PIN);

  if (isnan(temp) || isnan(hum))
  {
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("DHT ERROR!");
    delay(1000);
    return;
  }

  // Serial Monitor
  Serial.println("----------------------");
  Serial.print("Temperature : ");
  Serial.print(temp);
  Serial.println(" C");

  Serial.print("Humidity    : ");
  Serial.print(hum);
  Serial.println(" %");

  Serial.print("Gas Value   : ");
  Serial.println(gas);

  // Normal LCD Display
  lcd.setCursor(0,0);
  lcd.print("T:");
  lcd.print(temp,1);
  lcd.print((char)223);
  lcd.print("C ");

  lcd.print("H:");
  lcd.print(hum,0);
  lcd.print("% ");

  lcd.setCursor(0,1);
  lcd.print("Gas:");
  lcd.print(gas);
  lcd.print("    ");

  // Air Quality & LEDs

  // GOOD
  if(gas <= 800)
  {
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(YELLOW_LED, LOW);
    digitalWrite(RED_LED, LOW);
    digitalWrite(BUZZER, LOW);
  }

  // MODERATE
  else if(gas <= 1500)
  {
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(YELLOW_LED, HIGH);
    digitalWrite(RED_LED, LOW);
    digitalWrite(BUZZER, LOW);
  }

  // POOR
  else
  {
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(YELLOW_LED, LOW);
    digitalWrite(RED_LED, HIGH);
    digitalWrite(BUZZER, HIGH);

    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("***WARNING***");
    lcd.setCursor(0,1);
    lcd.print("HIGH POLLUTION");
    delay(1000);
    lcd.clear();
  }

  delay(1000);
}

Project Mission & Methodology

Background research on air quality monitoring, target SDGs, and smart IoT architecture.

Why Smart Air Quality Detection Matters

Air pollution is one of the leading environmental health risks worldwide. Harmful airborne pollutants such as Carbon Dioxide (CO2), Ammonia (NH3), Nitrous Oxides (NOx), and particulate smoke often accumulate silently in indoor classrooms, laboratories, and urban environments.

Our project combines low-cost high-precision sensors—the MQ135 Gas Sensor for chemical gas concentrations and the DHT11 Sensor for ambient thermal conditions—with an ESP32 Microcontroller to continuously transmit real-time air quality metrics to a centralized monitoring hub.

Key Features

Core functional capabilities of the Areosence monitoring system:

Responsive and user-friendly interface accessible from desktops, tablets, and smartphones.
Displays real-time ambient temperature telemetry readings (°C).
Displays humidity measurements (% RH) for ambient climate assessment.
Displays MQ135 gas sensor values (PPM) for harmful air contaminants.
Indicates air quality status into Good (≤800), Moderate (801-1500), or Poor (>1500).
Provides a comprehensive overview of the Areosence system and its end-to-end functionality.

4-Layer System Architecture

End-to-end telemetry processing pipeline:

Layer 1
Input Layer
Consists of the MQ135 sensor for detecting air contaminants and gas concentrations, paired with the DHT11 sensor to monitor ambient temperature and relative humidity levels.
Layer 2
Processing Layer
The ESP32 microcontroller processes raw sensor telemetry, applies calibration algorithms to calculate environmental metrics, and evaluates threshold values to classify air quality status into Good, Moderate, or Poor.
Layer 3
Communication Layer
Utilizes the ESP32’s integrated Wi-Fi module to securely transmit real-time sensor readings over the network to the central web platform.
Layer 4
Output / Presentation Layer
Features the dedicated Areosence web platform, which receives the data stream, visualizes live temperature, humidity, and air quality metrics, and displays intuitive, color-coded health status indicators for end users.
